### 2.4 서버액션 `src/app/admin/ops-issues/actions.ts`
```ts
export async function updateOpsIssue(id: string, patch: {
  status?: 'open'|'acknowledged'|'in_progress'|'resolved'|'ignored'
  assignee?: string | null
  resolution_note?: string | null
}) {
  const admin = await requireAdmin(); if (!admin) return { ok:false, error:'권한 없음' }
  const update: Record<string, unknown> = { ...patch, updated_at: new Date().toISOString() }
  if (patch.status === 'resolved') update.resolved_at = new Date().toISOString()
  if (patch.status && patch.status !== 'resolved') update.resolved_at = null
  const { error } = await serviceClient().from('ops_issues').update(update).eq('id', id)
  if (error) return { ok:false, error: error.message }
  revalidatePath('/admin/ops-issues')
  return { ok:true }
}
```
- ⚠️ **탐지기(429)가 자동으로 덮지 않는 필드만 사람이 관리**(status/assignee/resolution_note). fingerprint·title 등 진단 필드는 탐지기 소관.
